## Create the client

```go
package ourpocket

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "net/http"
    "strings"
)

type Client struct {
    BaseURL    string
    APIKey     string
    HTTPClient *http.Client
}

type Envelope[T any] struct {
    Message string `json:"message"`
    Status  string `json:"status"`
    Data    T      `json:"data"`
}

type CreateWalletInput struct {
    UserID   string `json:"userId,omitempty"`
    Currency string `json:"currency,omitempty"`
}

func (c *Client) CreateWallet(ctx context.Context, input CreateWalletInput) (map[string]any, error) {
    payload, err := json.Marshal(input)
    if err != nil {
        return nil, err
    }

    req, err := http.NewRequestWithContext(
        ctx,
        http.MethodPost,
        strings.TrimRight(c.BaseURL, "/")+"/wallets/create",
        bytes.NewReader(payload),
    )
    if err != nil {
        return nil, err
    }

    req.Header.Set("Authorization", "Bearer "+c.APIKey)
    req.Header.Set("Content-Type", "application/json")

    response, err := c.HTTPClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer response.Body.Close()

    if response.StatusCode < 200 || response.StatusCode >= 300 {
        return nil, fmt.Errorf("OurPocket returned %s", response.Status)
    }

    var result Envelope[map[string]any]
    if err := json.NewDecoder(response.Body).Decode(&result); err != nil {
        return nil, err
    }

    return result.Data, nil
}
```

## Use the client

```go
client := &ourpocket.Client{
    BaseURL: os.Getenv("OURPOCKET_BASE_URL"),
    APIKey: os.Getenv("OURPOCKET_API_KEY"),
    HTTPClient: &http.Client{Timeout: 30 * time.Second},
}

wallet, err := client.CreateWallet(ctx, ourpocket.CreateWalletInput{
    UserID: "customer_8462",
    Currency: "NGN",
})
```
