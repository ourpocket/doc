title: "TypeScript client"
description: "Create a typed OurPocket REST client with native fetch."
---

## Create the client

```ts
interface OurPocketConfig {
  baseUrl: string;
  apiKey: string;
}

interface CreateWalletInput {
  currency?: string;
  userId?: string;
}

export class OurPocketClient {
  constructor(private readonly config: OurPocketConfig) {}

  private async request<T>(path: string, init: RequestInit = {}): Promise<T> {
    const response = await fetch(`${this.config.baseUrl}${path}`, {
      ...init,
      headers: {
        Authorization: `Bearer ${this.config.apiKey}`,
        "Content-Type": "application/json",
        ...init.headers,
      },
    });

    const body = await response.json();

    if (!response.ok) {
      const message = Array.isArray(body.message)
        ? body.message.join(", ")
        : body.message ?? "OurPocket request failed";
      throw new Error(message);
    }

    return body.data as T;
  }

  createWallet(input: CreateWalletInput) {
    return this.request("/wallets/create", {
      method: "POST",
      body: JSON.stringify(input),
    });
  }

  getWallet(walletId: string) {
    return this.request(`/wallets/${walletId}`);
  }
}
```

## Use the client

```ts
const pocket = new OurPocketClient({
  baseUrl: process.env.OURPOCKET_BASE_URL!,
  apiKey: process.env.OURPOCKET_API_KEY!,
});

const wallet = await pocket.createWallet({
  userId: "customer_8462",
  currency: "NGN",
});
```

## Add a durable transaction

```ts
const transaction = await fetch(
  `${process.env.OURPOCKET_BASE_URL}/transactions`,
  {
    method: "POST",
    headers: {
      Authorization: `Bearer ${process.env.OURPOCKET_API_KEY}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      type: "credit",
      walletId: "wallet_id",
      amount: "5000",
      currency: "NGN",
      reference: "deposit_1284",
    }),
  },
).then((response) => response.json());
```
